% title: 포물선 운동
% description: 원점에서 속력 v₀, 각 θ로 던진 물체의 포물선 궤적과 초기 속도 성분, 최고점 높이 H, 수평 도달거리 R
% packages:
% libraries: arrows.meta
\begin{tikzpicture}[
    vec/.style={very thick, -{Stealth[length=3mm]}},
    comp/.style={thick, blue!70!black, -{Stealth[length=2.5mm]}}]
  \def\ang{50}     % 던진 각 θ (도)
  \def\R{8}        % 수평 도달거리 R (그림 길이)
  \def\vlen{2.2}   % 초기 속도 화살표 길이
  \pgfmathsetmacro\Hmax{\R*tan(\ang)/4}   % 최고점 높이 H = (R/4) tan θ
  % 좌표축
  \draw[-{Stealth}] (-0.3,0) -- (\R+1,0) node[right] {$x$};
  \draw[-{Stealth}] (0,-0.3) -- (0,\Hmax+1.3) node[above] {$y$};
  \node[below left] at (0,0) {$O$};
  % 궤적 y = x tan θ (1 - x/R)
  \draw[thick, domain=0:\R, samples=60] plot (\x, {\x*tan(\ang)*(1-\x/\R)});
  % 초기 속도와 성분
  \draw[comp] (0,0) -- ({\vlen*cos(\ang)},0) node[below] {$v_0\cos\theta$};
  \draw[comp] (0,0) -- (0,{\vlen*sin(\ang)}) node[left] {$v_0\sin\theta$};
  \draw[thin, dashed] ({\vlen*cos(\ang)},0) -- (\ang:\vlen) -- (0,{\vlen*sin(\ang)});
  \draw[vec] (0,0) -- (\ang:\vlen) node[above left] {$\vec{v}_0$};
  \draw (0.7,0) arc[start angle=0, end angle=\ang, radius=0.7];
  \node at (\ang/2:0.95) {$\theta$};
  % 최고점: 속도는 수평(v₀ cos θ), 높이 H
  \coordinate (P) at (\R/2,\Hmax);
  \draw[thin, dashed] (0,\Hmax) node[left] {$H$} -- (P) -- (\R/2,0);
  \fill (P) circle[radius=0.07];
  \node[above left] at (P) {최고점};
  \draw[vec] (P) -- ++({\vlen*cos(\ang)},0) node[above] {$v_0\cos\theta$};
  % 수평 도달거리
  \draw[{Stealth[length=2mm]}-{Stealth[length=2mm]}] (0,-0.9) -- (\R,-0.9)
    node[midway, fill=white] {$R$};
  \draw[thin] (\R,0) -- (\R,-1.1);
  \draw[thin] (0,-0.3) -- (0,-1.1);
\end{tikzpicture}
