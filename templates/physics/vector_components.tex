% title: 벡터의 성분 분해
% description: 좌표평면에서 x축과 각 θ를 이루는 벡터 A를 x성분 A cos θ와 y성분 A sin θ로 나눈 그림
% packages:
% libraries: arrows.meta
\begin{tikzpicture}[
    vec/.style={very thick, -{Stealth[length=3mm]}},
    comp/.style={thick, blue!70!black, -{Stealth[length=2.5mm]}}]
  \def\ang{35}    % 벡터가 x축과 이루는 각 θ (도)
  \def\A{4}       % 벡터의 크기 (화살표 길이)
  \coordinate (O) at (0,0);
  \coordinate (P) at (\ang:\A);    % 벡터의 끝
  % 좌표축
  \draw[-{Stealth}] (-0.5,0) -- ({\A*cos(\ang)+1},0) node[right] {$x$};
  \draw[-{Stealth}] (0,-0.5) -- (0,{\A*sin(\ang)+1}) node[above] {$y$};
  \node[below left] at (O) {$O$};
  % 성분과 투영선
  \draw[thin, dashed] (P) -- (P |- O);
  \draw[thin, dashed] (P) -- (P -| O);
  \draw[comp] (O) -- (P |- O) node[midway, below] {$A_x = A\cos\theta$};
  \draw[comp] (O) -- (P -| O) node[midway, left] {$A_y = A\sin\theta$};
  % 벡터와 각
  \draw[vec] (O) -- (P) node[midway, above left] {$\vec{A}$};
  \draw (1,0) arc[start angle=0, end angle=\ang, radius=1];
  \node at (\ang/2:1.3) {$\theta$};
\end{tikzpicture}
