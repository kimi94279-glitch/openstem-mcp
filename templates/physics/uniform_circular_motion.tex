% title: 등속 원운동
% description: 반지름 r인 원 위를 시계 반대 방향으로 도는 물체의 접선 방향 속도 v와 중심을 향하는 구심 가속도 a_c
% packages:
% libraries: arrows.meta
\begin{tikzpicture}[
    vec/.style={very thick, -{Stealth[length=3mm]}}]
  \def\rad{2.5}   % 원의 반지름 r
  \def\pos{40}    % 물체의 위치각 (도)
  \def\vlen{2}    % 속도 화살표 길이
  \def\alen{1.3}  % 구심 가속도 화살표 길이
  \coordinate (O) at (0,0);
  \coordinate (P) at (\pos:\rad);   % 물체의 위치
  \draw[thick] (O) circle[radius=\rad];
  \fill (O) circle[radius=0.05] node[below left] {$O$};
  \draw[thin, dashed] (O) -- (P) node[pos=0.25, below right] {$r$};
  % 운동 방향 (시계 반대 방향)
  \draw[-{Stealth}, gray] (200:\rad+0.35) arc[start angle=200, end angle=250, radius=\rad+0.35];
  % 속도: 접선 방향 / 구심 가속도: 중심 방향 (구심력 F = m v^2 / r도 같은 방향)
  \draw[vec] (P) -- ++(\pos+90:\vlen) node[above] {$\vec{v}$};
  \draw[vec, red!70!black] (P) -- ++(\pos+180:\alen) node[above left] {$\vec{a}_c$};
  \fill[blue!60!black] (P) circle[radius=0.15];
\end{tikzpicture}
