% packages: circuitikz

\begin{circuitikz}
  \draw (0,0) to[battery1, l=$\mathcal{E}_1$, invert] (0,3)
        to[R, l=$R_1$, i>_=$I_1$] (4,3)
        to[R, l=$R_3$, i>_=$I_3$, *-*] (4,0)
        -- (0,0);
  \draw (8,0) to[battery1, l_=$\mathcal{E}_2$, invert] (8,3)
        to[R, l_=$R_2$, i>^=$I_2$] (4,3);
  \draw (8,0) -- (4,0);
